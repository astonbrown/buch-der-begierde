   508→   508→function PassageCard({
   509→   509→  passage,
   510→   510→  index,

   513→   513→}: {

   542→   542→        borderColor: style.border,
   543→   543→      }}

   708→   708→export default function BuchDerBegierde() {
   709→   709→  const [activeChapter, setActiveChapter] = useState(0);
   710→   710→  const [showOverlay, setShowOverlay] = useState(false);
   711→   711→  const [overlayPhase, setOverlayPhase] = useState<'in' | 'out' | null>(null);
   712→   712→  const mountedRef = useRef(false);
   713→   713→
   714→   714→  useEffect(() => {
   715→   715→    mountedRef.current = true;
   716→   716→  }, []);
   717→   717→
   718→   718→  const navigateToChapter = useCallback((index: number) => {
   719→   719→    if (!mountedRef.current || overlayPhase || index === activeChapter) return;
   720→   720→    setOverlayPhase('in');
   721→   721→    setShowOverlay(true);
   722→   722→    setTimeout(() => {
   723→   723→      setActiveChapter(index);
   724→   724→      setOverlayPhase('out');
   725→   725→    }, 800);
   726→   726→    setTimeout(() => {
   727→   727→      setShowOverlay(false);
   728→   728→      setOverlayPhase(null);
   729→   729→    }, 1600);
   730→   730→  }, [overlayPhase, activeChapter]);
   731→   731→
   732→   732→  const handleEnter = useCallback(() => navigateToChapter(1), [navigateToChapter]);
   733→   733→  const handleContinue = useCallback(() => navigateToChapter(2), [navigateToChapter]);
   734→   734→
   735→   735→  return (
   736→   736→    <main style={{ background: '#C4A574', minHeight: '100vh' }}>
   737→   737→      {activeChapter === 0 && <ChapterIntro onEnter={handleEnter} />}
   738→   738→      {activeChapter === 1 && <ChapterBookIntro onContinue={handleContinue} />}
   739→   739→      {activeChapter === 2 && <ChapterPassages />}
   740→   740→
   741→   741→      {/* Transition overlay */}
   742→   742→      <TransitionOverlay phase={overlayPhase} />
   743→   743→
   744→   744→      {/* Nav dots */}
   745→   745→      {activeChapter > 0 && !overlayPhase && (
   746→   746→        <NavDots current={activeChapter - 1} onNavigate={navigateToChapter} />
   747→   747→      )}
   748→   748→    </main>
   749→   749→  );
   750→   750→}

   460→/* ======================================================== */
   461→/*  PASSAGE CARD (light parchment)                            */
   462→/* ======================================================== */
   463→
   464→function PassageCard({
   465→  passage,
   466→  index,
   467→  selectedMood,
   468→  onMoodSelect,
   469→}: {
   470→  passage: (typeof textPassages)[0];
   471→  index: number;
   472→  selectedMood: MoodType | null;
   473→  onMoodSelect: (id: number, mood: MoodType) => void;
   474→}) {
   475→  const [revealed, setRevealed] = useState(false);
   476→  const cardRef = useRef<HTMLDivElement>(null);
   477→
   478→  useEffect(() => {
   479→    const observer = new IntersectionObserver(
   480→      ([entry]) => {
   481→        if (entry.isIntersecting) setTimeout(() => setRevealed(true), index * 250);
   482→      },
   483→      { threshold: 0.15 }
   484→    );
   485→    if (cardRef.current) observer.observe(cardRef.current);
   486→    return () => observer.disconnect();
   487→  }, [index]);
   488→
   489→  const currentMood = selectedMood || passage.defaultMood;
   490→  const style = moodStyles[currentMood];
   491→
   492→  return (
   493→    <motion.div
   494→      ref={cardRef}
   495→      className="mood-card-light"
   496→      style={{
   497→        background: style.bg,
   498→        borderColor: style.border,
   499→      }}
   500→      initial={{ opacity: 0, y: 25 }}
   501→      animate={revealed ? { opacity: 1, y: 0 } : { opacity: 0, y: 25 }}
   502→      transition={{ duration: 1.2, ease: 'easeOut' }}
   503→    >
   504→      {/* Subtle glow at top */}
   505→      <div
   506→        className="absolute top-0 left-0 right-0 h-20 pointer-events-none transition-all duration-1000"
   507→        style={{ background: `linear-gradient(to bottom, ${style.glow}, transparent)` }}
   508→      />
   509→
   510→      {/* Quote mark */}
   511→      <div
   512→        className="font-[family-name:var(--font-cinzel)] text-4xl sm:text-5xl leading-none mb-2 select-none"
   513→        style={{ color: style.accent, opacity: 0.6 }}
   514→      >
   515→        „
   516→      </div>
   517→
   518→      {/* Text */}
   519→      <p
   520→        className="font-[family-name:var(--font-cormorant)] text-xl sm:text-2xl md:text-3xl leading-relaxed italic mb-5 transition-colors duration-1000"
   521→        style={{
   522→          color: style.text,
   523→          textShadow: '0 1px 1px rgba(255,245,220,0.25)',
   524→        }}

   645→
   646→export default function BuchDerBegierde() {
   647→  const [activeChapter, setActiveChapter] = useState(0);
   648→  const [transitioning, setTransitioning] = useState(false);
   649→
   650→  const navigateToChapter = useCallback((index: number) => {
   651→    if (transitioning) return;
   652→    setTransitioning(true);
   653→    setActiveChapter(index);
   654→    setTimeout(() => setTransitioning(false), 1200);
   655→  }, [transitioning]);
   656→
   657→  const handleEnter = useCallback(() => navigateToChapter(1), [navigateToChapter]);
   658→  const handleContinue = useCallback(() => navigateToChapter(2), [navigateToChapter]);
   659→
   660→  return (
   661→    <main className="relative min-h-screen" style={{ background: '#C4A574' }}>
   662→      <AnimatePresence mode="wait">
   663→        {activeChapter === 0 && (
   664→          <motion.div
   665→            key="chapter-1"
   666→            initial={{ opacity: 1 }}
   667→            exit={{ opacity: 0 }}
   668→            transition={{ duration: 1.5, ease: 'easeInOut' }}
   669→          >
   670→            <ChapterIntro onEnter={handleEnter} />
   671→          </motion.div>
   672→        )}
   673→
   674→        {activeChapter === 1 && (
   675→          <motion.div
   676→            key="chapter-2"
   677→            initial={{ opacity: 0 }}
   678→            animate={{ opacity: 1 }}
   679→            exit={{ opacity: 0 }}
   680→            transition={{ duration: 1.5, ease: 'easeInOut' }}
   681→          >
   682→            <ChapterBookIntro onContinue={handleContinue} />
   683→          </motion.div>
   684→        )}
   685→
   686→        {activeChapter === 2 && (
   687→          <motion.div
   688→            key="chapter-3"
   689→            initial={{ opacity: 0 }}
   690→            animate={{ opacity: 1 }}
   691→            transition={{ duration: 1.5, ease: 'easeInOut' }}
   692→          >
   693→            <ChapterPassages />
   694→          </motion.div>
   695→        )}
   696→      </AnimatePresence>
   697→
   698→      {activeChapter > 0 && (
   699→        <NavDots current={activeChapter - 1} onNavigate={navigateToChapter} />
   700→      )}
   701→    </main>
   702→  );
   703→}

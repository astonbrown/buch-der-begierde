   504→/* ======================================================== */
   505→/*  PASSAGE CARD (light parchment)                            */
   506→/* ======================================================== */
   507→
   508→function PassageCard({
   509→  passage,
   510→  index,
   511→  selectedMood,
   512→  onMoodSelect,
   513→}: {
   514→  passage: (typeof textPassages)[0];
   515→  index: number;
   516→  selectedMood: MoodType | null;
   517→  onMoodSelect: (id: number, mood: MoodType) => void;
   518→}) {
   519→  const [revealed, setRevealed] = useState(false);
   520→  const cardRef = useRef<HTMLDivElement>(null);
   521→
   522→  useEffect(() => {
   523→    const observer = new IntersectionObserver(
   524→      ([entry]) => {
   525→        if (entry.isIntersecting) setTimeout(() => setRevealed(true), index * 250);
   526→      },
   527→      { threshold: 0.15 }
   528→    );
   529→    if (cardRef.current) observer.observe(cardRef.current);
   530→    return () => observer.disconnect();
   531→  }, [index]);
   532→
   533→  const currentMood = selectedMood || passage.defaultMood;
   534→  const style = moodStyles[currentMood];
   535→

   686→function TransitionOverlay({ phase }: { phase: 'in' | 'out' | null }) {
   687→  if (!phase || typeof document === 'undefined') return null;
   688→  return createPortal(
   689→    <motion.div
   690→      className="fixed inset-0"
   691→      style={{
   692→        zIndex: 99998,
   693→        background: 'radial-gradient(ellipse at center, #A68B5B 0%, #8B7355 50%, #6B563F 100%)',
   694→        pointerEvents: 'none',
   695→      }}
   696→      initial={{ opacity: 0 }}
   697→      animate={{ opacity: phase === 'in' ? 1 : 0 }}
   698→      transition={{ duration: 0.8, ease: 'easeInOut' }}
   699→    />,
   700→    document.body
   701→  );
   702→}

   708→export default function BuchDerBegierde() {
   709→  const [activeChapter, setActiveChapter] = useState(0);
   710→  const [showOverlay, setShowOverlay] = useState(false);
   711→  const [overlayPhase, setOverlayPhase] = useState<'in' | 'out' | null>(null);
   712→  const mountedRef = useRef(false);
   713→
   714→  useEffect(() => {
   715→    mountedRef.current = true;
   716→  }, []);
   717→
   718→  const navigateToChapter = useCallback((index: number) => {
   719→    if (!mountedRef.current || overlayPhase || index === activeChapter) return;
   720→    setOverlayPhase('in');
   721→    setShowOverlay(true);
   722→    setTimeout(() => {
   723→      setActiveChapter(index);
   724→      setOverlayPhase('out');
   725→    }, 800);
   726→    setTimeout(() => {
   727→      setShowOverlay(false);
   728→      setOverlayPhase(null);
   729→    }, 1600);
   730→  }, [overlayPhase, activeChapter]);
   731→
   732→  const handleEnter = useCallback(() => navigateToChapter(1), [navigateToChapter]);
   733→  const handleContinue = useCallback(() => navigateToChapter(2), [navigateToChapter]);
   734→
   735→  return (
   736→    <main style={{ background: '#C4A574', minHeight: '100vh' }}>
   737→      {activeChapter === 0 && <ChapterIntro onEnter={handleEnter} />}
   738→      {activeChapter === 1 && <ChapterBookIntro onContinue={handleContinue} />}
   739→      {activeChapter === 2 && <ChapterPassages />}
   740→
   741→      {/* Transition overlay */}
   742→      <TransitionOverlay phase={overlayPhase} />
   743→
   744→      {/* Nav dots */}
   745→      {activeChapter > 0 && !overlayPhase && (
   746→        <NavDots current={activeChapter - 1} onNavigate={navigateToChapter} />
   747→      )}
   748→    </main>
   749→  );
   750→}

   855→function wrapWords(text) {
   856→  return text.split(/(\s+)/).map(token => {
   857→    if (/^\s+$/.test(token)) return token;
   858→    return `<span class="word">${token}</span>`;
   859→  }).join('');
   860→}
   861→
   862→function revealWords(container, delayPerWord = 70) {
   863→  const words = container.querySelectorAll('.word');
   864→  words.forEach((w, i) => {
   865→    setTimeout(() => w.classList.add('visible'), i * delayPerWord);
   866→  });
   867→  return words.length * delayPerWord;
   868→}

   870→function runPortalAnimation() {
   871→  if (portalAnimated) return;
   872→  portalAnimated = true;
   873→
   874→  const titleEl = document.getElementById('portalTitle');
   875→  const textLeft = document.getElementById('portalTextLeft');
   876→  const textRight = document.getElementById('portalTextRight');
   877→  const divider = document.getElementById('portalDivider');
   878→  const codeBox = document.getElementById('portalCode');
   879→  const hint = document.getElementById('portalHint');
   880→  const btn = document.getElementById('portalBtn');
   881→
   882→  // --- LEFT PAGE: title + first half ---
   883→  titleEl.innerHTML = wrapWords(PORTAL_TITLE);
   884→  let delay = revealWords(titleEl, 75);
   885→
   886→  setTimeout(() => {
   887→    divider.style.transition = 'opacity 0.7s ease';
   888→    divider.style.opacity = '0.7';
   889→  }, delay + 150);
   890→
   891→  let leftDelay = delay + 400;
   892→  PORTAL_LEFT.forEach((para) => {
   893→    const p = document.createElement('p');
   894→    p.innerHTML = wrapWords(para);
   895→    textLeft.appendChild(p);
   896→    setTimeout(() => revealWords(p, 50), leftDelay);
   897→    leftDelay += para.split(/\s+/).length * 50 + 350;
   898→  });
   899→
   900→  // --- RIGHT PAGE: second half + controls ---
   901→  // Start right page a bit after left has begun, so reading flows left → right
   902→  let rightDelay = Math.max(leftDelay * 0.45, 1200);
   903→  PORTAL_RIGHT.forEach((para) => {
   904→    const p = document.createElement('p');
   905→    p.innerHTML = wrapWords(para);
   906→    textRight.appendChild(p);
   907→    setTimeout(() => revealWords(p, 50), rightDelay);
   908→    rightDelay += para.split(/\s+/).length * 50 + 350;
   909→  });
   910→
   911→  setTimeout(() => {
   912→    codeBox.style.transition = 'opacity 0.8s ease';
   913→    codeBox.style.opacity = '1';
   914→    codeBox.style.pointerEvents = 'auto';
   915→    hint.style.transition = 'opacity 0.8s ease';

   917→    btn.style.transition = 'opacity 0.8s ease';
   918→    btn.style.opacity = '1';
   919→    btn.style.pointerEvents = 'auto';
   920→  }, rightDelay + 250);
   921→}

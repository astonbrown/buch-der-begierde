   744→function resizeBookCanvas() {

   751→let bookAnimFrame = 0;
   752→const bookDuration = 180; // frames (~3s bei 60fps)
   753→
   754→function drawBookOpen(progress) {
   755→  const w = bookCanvas.width, h = bookCanvas.height;
   756→  bookCtx.fillStyle = '#0a0a0f';
   757→  bookCtx.fillRect(0, 0, w, h);
   758→
   759→  const cx = w / 2, cy = h / 2;
   760→  const bookW = Math.min(w * 0.5, 500);
   761→  const bookH = Math.min(h * 0.55, 400);
   762→  const spineX = cx;
   763→  const coverAngle = progress * Math.PI * 0.45; // max 81°
   764→  const pageAngle = progress * Math.PI * 0.35;
   765→
   766→  // Schatten
   767→  bookCtx.save();
   768→  bookCtx.shadowColor = 'rgba(0,0,0,0.6)';
   769→  bookCtx.shadowBlur = 40;
   770→  bookCtx.shadowOffsetY = 20;
   771→
   772→  // Linke Seite (Cover öffnet sich)
   773→  bookCtx.save();
   774→  bookCtx.translate(spineX, cy);
   775→  bookCtx.scale(Math.cos(coverAngle), 1);
   776→  bookCtx.fillStyle = '#2a1f15';
   777→  bookCtx.fillRect(-bookW/2, -bookH/2, bookW/2, bookH);
   778→  // Cover-Details
   779→  bookCtx.strokeStyle = '#4a3a28';
   780→  bookCtx.lineWidth = 2;
   781→  bookCtx.strokeRect(-bookW/2 + 8, -bookH/2 + 8, bookW/2 - 16, bookH - 16);
   782→  bookCtx.restore();
   783→
   784→  // Rechte Seite (Cover öffnet sich)
   785→  bookCtx.save();
   786→  bookCtx.translate(spineX, cy);
   787→  bookCtx.scale(Math.cos(coverAngle), 1);
   788→  bookCtx.fillStyle = '#2a1f15';
   789→  bookCtx.fillRect(0, -bookH/2, bookW/2, bookH);
   790→  bookCtx.strokeStyle = '#4a3a28';
   791→  bookCtx.lineWidth = 2;
   792→  bookCtx.strokeRect(8, -bookH/2 + 8, bookW/2 - 16, bookH - 16);
   793→  bookCtx.restore();
   794→
   795→  bookCtx.restore();
   796→
   797→  // Linke Seite (Seiten)
   798→  bookCtx.save();
   799→  bookCtx.translate(spineX, cy);
   800→  bookCtx.scale(Math.cos(pageAngle), 1);
   801→  bookCtx.fillStyle = '#e8dcc8';
   802→  bookCtx.fillRect(-bookW/2 + 4, -bookH/2 + 4, bookW/2 - 8, bookH - 8);
   803→  // Seiten-Textur
   804→  bookCtx.fillStyle = 'rgba(180,170,150,0.1)';
   805→  for (let i = 0; i < 20; i++) {
   806→    bookCtx.fillRect(-bookW/2 + 10, -bookH/2 + 15 + i * (bookH-30)/20, bookW/2 - 20, 1);
   807→  }
   808→  bookCtx.restore();
   809→
   810→  // Rechte Seite (Seiten)
   811→  bookCtx.save();
   812→  bookCtx.translate(spineX, cy);
   813→  bookCtx.scale(Math.cos(pageAngle), 1);
   814→  bookCtx.fillStyle = '#e8dcc8';
   815→  bookCtx.fillRect(4, -bookH/2 + 4, bookW/2 - 8, bookH - 8);
   816→  bookCtx.fillStyle = 'rgba(180,170,150,0.1)';
   817→  for (let i = 0; i < 20; i++) {
   818→    bookCtx.fillRect(10, -bookH/2 + 15 + i * (bookH-30)/20, bookW/2 - 20, 1);
   819→  }
   820→  bookCtx.restore();
   821→
   822→  // Buchrücken
   823→  bookCtx.fillStyle = '#1a1410';
   824→  bookCtx.fillRect(spineX - 3, cy - bookH/2, 6, bookH);
   825→
   826→  // Partikel
   827→  if (progress > 0.3) {
   828→    const pCount = Math.floor((progress - 0.3) * 100);
   829→    for (let i = 0; i < pCount; i++) {
   830→      const px = cx + (Math.random() - 0.5) * bookW * 1.5;
   831→      const py = cy + (Math.random() - 0.5) * bookH * 1.2;
   832→      const ps = Math.random() * 1.5 + 0.3;
   833→      const po = (progress - 0.3) * 0.4 * Math.random();
   834→      bookCtx.beginPath();
   835→      bookCtx.arc(px, py, ps, 0, Math.PI * 2);
   836→      bookCtx.fillStyle = `rgba(184,160,96,${po})`;
   837→      bookCtx.fill();
   838→    }
   839→  }
   840→}

   842→function animateBook() {
   843→  if (bookAnimFrame >= bookDuration) {
   844→    bookOpening.classList.remove('active');
   845→    showPortal();
   846→    return;
   847→  }
   848→  const progress = bookAnimFrame / bookDuration;
   849→  const eased = 1 - Math.pow(1 - progress, 3); // ease-out cubic
   850→  drawBookOpen(eased);
   851→  bookAnimFrame++;
   852→  requestAnimationFrame(animateBook);
   853→}
   854→
   855→function startBookOpening() {
   856→  bookOpening.classList.add('active');
   857→  bookAnimFrame = 0;
   858→  animateBook();
